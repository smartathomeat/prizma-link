#include <tunables/global>

profile prizma_link flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  file,
  signal (send) set=(kill,term,int,hup,cont),

  # Official Home Assistant s6-overlay minimum.
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio loads its functions from this directory.
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  /opt/prizma-link/** r,
  /data/** rw,
  /run/dbus/system_bus_socket rw,
  /etc/hosts r,
  /etc/resolv.conf r,

  network inet stream,
  network inet6 stream,
  network bluetooth stream,

  dbus (send, receive)
       bus=system
       peer=(name=org.bluez),
  dbus (bind)
       bus=system
       name=io.smartathome.prizma_link,

  deny /var/run/docker.sock rw,
}
